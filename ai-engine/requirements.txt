# AI Video Intelligence Phase 2 note: this used to be opencv-python-headless.
# ultralytics depends on opencv-python (the GUI-capable build) by package name, and
# pip has no notion that opencv-python-headless satisfies it — installing both in the
# same environment silently corrupts the shared cv2 native module (discovered for
# real: cv2.CascadeClassifier disappeared after adding ultralytics). Switched to the
# non-headless build at the same pinned version instead; this codebase never calls a
# GUI function (cv2.imshow etc.) so behavior is unchanged, and the Dockerfile already
# installs libgl1/libglib2.0-0, which opencv-python needs to import at all.
opencv-python==4.10.0.84
numpy==2.1.3
httpx==0.28.1
pydantic-settings==2.7.0
# AI Video Intelligence Phase 2 — real multi-class detection (YOLOv8n). Pulls in
# torch as a transitive dependency — a real, disclosed Docker image size increase,
# applied on every build regardless of whether any camera opts in
# (Camera.multi_class_detection_enabled) since Docker images are static. See
# CLAUDE.md "Known limitations" and app/detectors/yolo_detector.py.
#
# Real bug found and fixed: on Windows, plain `pip install torch` resolves to a
# CPU-only wheel (~311MB, measured locally) — but on Linux (the actual deployment
# target, per the Dockerfile), PyPI's default torch wheel bundles full CUDA runtime
# libraries (nvidia-cudnn/cublas/etc.), several GB uncompressed, even though this
# platform only ever runs AI_DEVICE=cpu inference and has no GPU. Deploying this to
# a real VM filled its disk mid-build ("no space left on device") purely from this —
# not from anything AI Video Intelligence actually needs. Forcing the CPU-only wheel
# index (same one the official PyTorch install docs recommend for CPU-only setups)
# fixes this for real on Linux while remaining a no-op on Windows dev machines.
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.5.1
ultralytics==8.4.150
